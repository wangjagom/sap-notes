    "sap/ui/core/mvc/Controller",
    "sap/ui/model/json/JSONModel",
    "sap/ui/model/Filter",
    "sap/ui/model/FilterOperator"
], function (Controller, JSONModel, Filter, FilterOperator) {
    "use strict";

    return Controller.extend("ncode.zcXXw0205.controller.View1", {
        onInit: async function () {
            const oModel = new JSONModel();
            await oModel.loadData("model/data.json");      // 파일이 도착할 때까지 기다려요
            this.getView().setModel(oModel);               // 화면에 모델 연결
        },

        onSearch: function () {
            const sClass = this.byId("cmbClass").getSelectedKey();
            const oBinding = this.byId("tabStd").getBinding("items");
            const aFilter = sClass ? [new Filter("Cid", FilterOperator.EQ, sClass)] : [];
            oBinding.filter(aFilter);                      // 반을 안 골랐으면 전체 보기
        },

        onSelect: function (oEvent) {
            const oItem = oEvent.getParameter("listItem");           // 고른 줄
            const sPath = oItem.getBindingContext().getPath();       // 그 줄의 위치
            this.byId("pnlDetail").bindElement(sPath);               // 상세 패널의 기준으로 지정
        }
    });
});
